\begin{frame}{Cel, zakres i hipoteza}
  \begin{itemize}
    \item Cztery SZBD: 2 relacyjne (PostgreSQL 17, MariaDB 11.8) i~2 nierelacyjne
      (MongoDB 8.0, Valkey 8.1); trzy modele danych.
    \item Domena: rezerwacja biletów. Łączy transakcje, blokady, JSON i~duży
      ruch równoległy.
    \item 24 scenariusze CRUD (6 na operację) + 3 na danych JSON,
      \SizesList{} biletów, bez indeksów i~z~indeksami, po 5 prób.
    \item Test współbieżności: 1--1000 klientów rezerwuje te same 500 miejsc.
  \end{itemize}
  \vspace{2mm}
  \begin{block}{Hipoteza H1}
    Wraz ze wzrostem liczby równoległych klientów rezerwujących te same miejsca
    przewaga wydajnościowa Valkey i~MongoDB nad bazami relacyjnymi rośnie.
  \end{block}
  {\small Dodatkowo H2: różnice między silnikami rosną z~liczbą rekordów.}
\end{frame}

\begin{frame}{Model danych}
  \begin{columns}[T]
    \column{0.33\textwidth}
    \textbf{Relacyjny (12 tabel)}\\[1mm]
    \small obiekty, sektory, miejsca, wydarzenia, cenniki, kategorie,
    organizatorzy, klienci, kody\_rabatowe, rezerwacje, bilety, platnosci.
    Bilet to pula sprzedażowa: miejsce × wydarzenie ze statusem.
    \column{0.33\textwidth}
    \textbf{Dokumentowy}\\[1mm]
    \small wydarzenie z~osadzonym cennikiem, rezerwacja z~osadzonymi biletami
    i~płatnością, bilety jako osobna kolekcja (to je zmieniają równolegli
    klienci). Transakcje na replica secie.
    \column{0.33\textwidth}
    \textbf{Klucz--wartość}\\[1mm]
    \small hash wydarzenia, hash biletów, \texttt{SET} wolnych miejsc,
    rezerwacja jako JSON, liczniki w~\texttt{ZSET}, blokada miejsca z~TTL.
    ,,Indeksy'' to zbiory \texttt{ix:*} utrzymywane przez aplikację.
  \end{columns}
  \vspace{4mm}
  \small Dane z~deterministycznego generatora: te same wiersze w~każdej bazie,
  potwierdzone sumą kontrolną. Metadane wszędzie jako JSON (JSONB, JSON, BSON,
  JSON w~Valkey).
\end{frame}

\begin{frame}{Plany zapytań (EXPLAIN)}
  \small
  \begin{itemize}
    \item Pełne skany w~13 planach odczytu, 10 mln biletów: PostgreSQL
      \fullscans{postgres}{10000000}{off} $\to$ \fullscans{postgres}{10000000}{on},
      MariaDB \fullscans{mariadb}{10000000}{off} $\to$ \fullscans{mariadb}{10000000}{on},
      MongoDB \fullscans{mongo}{10000000}{off} $\to$ \fullscans{mongo}{10000000}{on}.
    \item R4 w~PostgreSQL: \texttt{Parallel Seq Scan} + \texttt{Hash Join}
      $\to$ dwa \texttt{Index Scan}; MariaDB: pełny skan \texttt{ALL}
      $\to$ dostęp przez indeks \texttt{ref}. Czas silnika: setki ms $\to$ dziesiąte części ms.
    \item R6 w~MongoDB zostaje \texttt{COLLSCAN} mimo indeksu: \texttt{\$match}
      po \texttt{status} nie pasuje do prefiksu \texttt{(wydarzenie\_id, status)}.
      PostgreSQL robi \texttt{Index Only Scan}.
    \item J1 na JSON: indeks GIN \texttt{jsonb\_path\_ops} (PostgreSQL),
      kolumna wirtualna (MariaDB), indeks na polach zagnieżdżonych (MongoDB).
  \end{itemize}
\end{frame}

\begin{frame}{Weryfikacja hipotez}
  \begin{block}{H1: \HOneVerdict{} (\HOneGrows/\HOneCases{} przypadków)}
    \small Przewaga Valkey nad PostgreSQL przy 10 mln:
    \cratio{valkey}{postgres}{10000000}{lua}{cond}{1}$\times$ (1 klient) $\to$
    \cratio{valkey}{postgres}{10000000}{lua}{cond}{1000}$\times$ (1000 klientów).
    MongoDB/PostgreSQL: \cratio{mongo}{postgres}{10000000}{atomic}{cond}{1} $\to$
    \cratio{mongo}{postgres}{10000000}{atomic}{cond}{1000}. Pozycja MongoDB rośnie,
    ale przewagę ma tylko nad MariaDB. Potwierdzona dla Valkey, częściowo dla MongoDB.
  \end{block}
  \begin{block}{H2: \HTwoVerdict{} (\HTwoGrows/\HTwoCases{})}
    \small Rozrzut rośnie, gdy bazy mają różną złożoność ścieżki dostępu
    (brak indeksu, osadzenie a~złączenie, licznik a~agregacja).
    Nie rośnie dla operacji po kluczu.
  \end{block}
  \small Zastrzeżenia: Valkey bez zapisu na dysk, jeden komputer, pojedyncze węzły bez TLS.
\end{frame}

\begin{frame}{Wnioski}
  \begin{enumerate}\small
    \item Nie ma jednej najszybszej bazy. Valkey wygrywa w~operacjach po kluczu
      i~przy współbieżności, ale bez struktur dostępu jest najwolniejszy.
    \item Indeksy zmieniają czas o~rzędy wielkości. Silnik po zaindeksowaniu
      zwykle o~mniej niż rząd.
    \item Model danych zmienia złożoność: osadzenie daje odczyt agregatu bez
      złączeń i~bez indeksu na kluczu obcym.
    \item Przy 1000 klientach strategia blokowania decyduje o~przepustowości.
    \item System biletowy: transakcje w~PostgreSQL, blokady miejsc z~TTL
      i~liczniki w~Valkey.
  \end{enumerate}
\end{frame}
